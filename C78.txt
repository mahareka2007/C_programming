#include <stdio.h>

int main() {
    int a;
    scanf("%d",&a);
    if(a==0){
        printf("Zero");
    }
    else if(a>0&&a%2==0){
        printf("Positive Even");
    }
    else if(a>0&&a%2!=0){
        printf("Positive Odd");
    }
    else if(a<0){
        printf("Negative");
    }
  

    return 0;
}
