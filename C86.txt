#include <stdio.h>

int main() {
    int a[5] = {10, 40, 20, 30, 50};
    int i, max, second;

    max = second = a[0];

    for(i = 1; i < 5; i++) {
        if(a[i] > max) {
            second = max;
            max = a[i];
        } else if(a[i] > second && a[i] != max) {
            second = a[i];
        }
    }

    printf("Second Largest = %d", second);
    return 0;
}
